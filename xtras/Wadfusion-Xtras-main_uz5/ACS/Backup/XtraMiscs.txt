function bool IsFloorLevel(int tag, int level)
{
    int flr = GetSectorFloorZ(tag,0,0) >> 16;
    if (flr == level)
        return 1;
    return 0;
}

function bool IsDoorOpen(int tag)
{
    int height = GetSectorCeilingZ(tag,0,0) >> 16;
    int flr = GetSectorFloorZ(tag,0,0) >> 16;
    
    if (height > flr)
        return 1;
    return 0;
}

function bool IsDoorClosed(int tag)
{
    return IsDoorOpen(tag) == 0;
}

// Weapon classes that only exist when the Unofficial Xtras addon is loaded.
// CheckInventory/GiveInventory/TakeInventory print "Unknown actor class" every tick for classes
// that don't exist, so these calls are skipped unless the addon is present.
int oeWeaponsLoaded = 0;

function int IsOEWeapon(str cls)
{
	return (StrICmp(cls, "H2P_RocketLauncher") == 0) ||
	       (StrICmp(cls, "H2P_PlasmaTrident") == 0) ||
	       (StrICmp(cls, "H2P_IncendiaryTrident") == 0) ||
	       (StrICmp(cls, "DCX_DZ0PlasmaRifle") == 0);
}

function int SafeCheck(str cls)
{
	if (IsOEWeapon(cls) && !oeWeaponsLoaded)
		return 0;
	return CheckInventory(cls);
}

function void SafeGive(str cls, int amount)
{
	if (IsOEWeapon(cls) && !oeWeaponsLoaded)
		return;
	GiveInventory(cls, amount);
}

function void SafeTake(str cls, int amount)
{
	if (IsOEWeapon(cls) && !oeWeaponsLoaded)
		return;
	TakeInventory(cls, amount);
}
